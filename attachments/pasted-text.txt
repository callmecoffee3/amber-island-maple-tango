        <button class="abtn" type="button" id="dc-cancel">Cancel</button>
      </div>`;
  } else {
    body.innerHTML = `
      <label>File name</label>
      <input type="text" id="dc-name" placeholder="notes.txt" />
      <label>Content (optional)</label>
      <textarea id="dc-content" placeholder="Write something…"></textarea>
      <div class="crow">
        <button class="abtn pri" type="button" id="dc-ok">Create file</button>
        <button class="abtn" type="button" id="dc-cancel">Cancel</button>
      </div>`;
  }
  el.classList.add("open");
  const input = body.querySelector("#dc-name");
  setTimeout(() => input && input.focus(), 40);
  body.querySelector("#dc-cancel").onclick = closeDeskCreate;
  body.querySelector("#dc-ok").onclick = () => {
    const name = (body.querySelector("#dc-name").value || "").trim();
    if (!name) { input && input.focus(); return; }
    const ch = ensure("Desktop");
    if (!ch) return;
    if (ch[name]) { alert("Already exists on desktop"); return; }
    if (kind === "folder") ch[name] = { type: "folder", children: {} };
    else ch[name] = { type: "file", content: (body.querySelector("#dc-content") && body.querySelector("#dc-content").value) || "" };
    paintIcons();
    closeDeskCreate();

  input && input.addEventListener("keydown", e => {
    if (e.key === "Enter") body.querySelector("#dc-ok").click();
    if (e.key === "Escape") closeDeskCreate();
  });
}
function addFolderAt(path) {
  if (path === "Desktop") { openDeskCreate("folder"); return true; }
  const name = window.prompt ? prompt("Folder name:") : null;
  if (!name || !name.trim()) return false;
  const ch = ensure(path);
  if (!ch) return false;
  const n = name.trim();
  if (ch[n]) { alert("Already exists"); return false; }
  ch[n] = { type: "folder", children: {} };
  return true;
}
function addFileAt(path) {
  if (path === "Desktop") { openDeskCreate("file"); return true; }
  const name = window.prompt ? prompt("File name (e.g. notes.txt):") : null;
  if (!name || !name.trim()) return false;
  const ch = ensure(path);
  if (!ch) return false;
  const n = name.trim();
  if (ch[n]) { alert("Already exists"); return false; }
  const content = (window.prompt ? prompt("Content (optional):") : "") || "";
  ch[n] = { type: "file", content };
  return true;
}

/* ═══════════ Desktop icons ═══════════ */
function paintIcons() {
  const desk = document.getElementById("desk");
  desk.querySelectorAll(".icon").forEach(e => e.remove());
  const skip = new Set(["Portrait"]);
  let row = 0, col = 0;
  Object.keys(kids("Desktop")).filter(n => !skip.has(n)).forEach(name => {
    const item = kids("Desktop")[name];
    const el = document.createElement("div");
    el.className = "icon";
    el.style.left = (20 + col * 98) + "px";
    el.style.top = (52 + row * 96) + "px";
    el.innerHTML = `<div class="g">${icon(name, item.type)}</div><div class="l">${esc(name)}</div>`;
    el.onclick = e => {
      e.stopPropagation();
      desk.querySelectorAll(".icon").forEach(i => i.classList.remove("sel"));
      el.classList.add("sel");
    };
    el.ondblclick = e => { e.stopPropagation(); openPath("Desktop/" + name); };
    desk.appendChild(el);
    row++;
    if (row > 5) { row = 0; col++; }
  });
}

/* ═══════════ Windows ═══════════ */
function openPath(path) {
  if (path === "Desktop/Power") {
    powerDown();
    return;
  }
  for (const [id, w] of wins) {
    if (w.path === path) { focus(id); return; }
  }
  const n = node(path);
  if (!n) return;
  const id = uid();
  const name = parts(path).pop();
  const folder = n.type === "folder";
  const pic = n.type === "pic";
  const win = document.createElement("div");
  win.className = "win on";
  win.dataset.id = id;
  win.style.width = folder ? "430px" : pic ? "340px" : "460px";
  win.style.height = folder ? "350px" : pic ? "390px" : "340px";
  win.style.left = (70 + wins.size * 24) + "px";
  win.style.top = (46 + wins.size * 20) + "px";
  win.style.zIndex = ++z;
  win.innerHTML = `
    <div class="tb">
      <span class="ic">${icon(name, n.type)}</span>
      <span class="tt">${esc(name)}</span>
      <div class="ctls">
        <button data-a="min">─</button>
        <button data-a="max">□</button>
        <button class="x" data-a="close">✕</button>
      </div>
    </div>
    <div class="body"></div>
    <div class="rh"></div>`;
  document.getElementById("desk").appendChild(win);
  wins.set(id, { el: win, path, type: n.type, min: false });
  active = id;
  const body = win.querySelector(".body");
  if (folder) paintFolder(body, path);
  else if (pic) paintPic(body);
  else paintFile(body, path, n);
  wire(win, id);
  paintBar();
}

function paintFolder(body, path) {
  // Build full folder tree from Desktop
  function collectTree(base, n, depth, out) {
    if (!n || n.type !== "folder") return;
    out.push({ path: base, name: parts(base).pop() || base, depth });
    Object.keys(n.children || {}).sort().forEach(k => {
      if (n.children[k].type === "folder") collectTree(base + "/" + k, n.children[k], depth + 1, out);
    });
  }
  const tree = [];
  collectTree("Desktop", FS.Desktop, 0, tree);

  const c = kids(path);
  const names = Object.keys(c).sort((a, b) => {
    const ta = c[a].type === "folder" ? 0 : 1;
    const tb = c[b].type === "folder" ? 0 : 1;
    if (ta !== tb) return ta - tb;
    return a.localeCompare(b);
  });

  let treeHtml = tree.map(item => {
    const on = item.path === path ? " on" : "";
    const pad = "&nbsp;".repeat(item.depth * 2);
    return `<button type="button" class="ti${on}" data-tree="${item.path}" title="${esc(item.path)}"><span class="pad">${pad}</span>📁 ${esc(item.name)}</button>`;
  }).join("");

  let crumb = `<div class="crumb">`;
  let acc = "";
  parts(path).forEach((p, i) => {
    acc += (acc ? "/" : "") + p;
    if (i) crumb += `<span class="s">/</span>`;
    crumb += `<button type="button" data-nav="${acc}">${esc(p)}</button>`;
  });
  crumb += `</div>`;

  let main = crumb + `
  <div class="fbar">
    <button class="pri" data-act="nf" type="button">+ Folder</button>
    <button data-act="nfile" type="button">+ File</button>
    <button data-act="up" type="button">Upload</button>
  </div>
  <div class="create-box hidden" id="create-box"></div>
  <div class="move-bar" id="move-bar">
    <span>Move selected to:</span>
    <select id="move-target"></select>
    <button class="abtn pri" type="button" id="move-go">Move</button>
    <button class="abtn" type="button" id="move-cancel">Cancel</button>
  </div>
  <p class="preview-note">Double-click folder to open here · Double-click file to read · Tree opens any folder</p>`;

  if (!names.length) {
    main += `<div class="empty"><span>📂</span>Empty — use + Folder or + File above</div>`;
  } else {
    main += `<div class="fgrid">`;
    names.forEach(name => {
      const attrName = String(name).replace(/&/g,"&amp;").replace(/"/g,"&quot;").replace(/</g,"&lt;");
      main += `<div class="fitem" data-open="${path}/${name}" data-name="${attrName}" data-type="${c[name].type}">
        <div class="acts">
          <button type="button" data-act="ren" title="Rename">✎</button>
          <button type="button" data-act="mov" title="Move">⇪</button>
          <button type="button" class="del" data-act="del" title="Delete">✕</button>
        </div>
        <div class="fg">${icon(name, c[name].type)}</div>
        <div class="fn">${esc(name)}</div>
      </div>`;
    });
    main += `</div>`;
  }

  body.innerHTML = `<div class="folder-layout">
    <div class="folder-tree">${treeHtml}</div>
    <div class="folder-main">${main}</div>
  </div>`;

  const mainEl = body.querySelector(".folder-main");
  const box = body.querySelector("#create-box");
  const moveBar = body.querySelector("#move-bar");
  let selectedName = null;

  function navTo(targetPath) {
    const winId = body.closest(".win").dataset.id;
    const w = wins.get(winId);
    if (!w) return;
    const n = node(targetPath);
    if (!n) return;
    w.path = targetPath;
    const name = parts(w.path).pop();
    w.el.querySelector(".tt").textContent = name;
    w.el.querySelector(".ic").textContent = icon(name, n.type);
    if (n.type === "folder") paintFolder(body, w.path);
    else if (n.type === "pic") paintPic(body);
    else paintFile(body, w.path, n);
    paintBar();
  }

  function showCreate(kind) {
    if (kind === "folder") {
      box.innerHTML = `
        <label>New folder name</label>
        <input type="text" id="cr-name" placeholder="Folder name" />
        <div class="crow">
          <button class="abtn pri" type="button" id="cr-ok">Create folder</button>
          <button class="abtn" type="button" id="cr-cancel">Cancel</button>
        </div>`;
    } else {
      box.innerHTML = `
        <label>New file name</label>
        <input type="text" id="cr-name" placeholder="notes.txt" />
        <label>Content (optional)</label>
        <textarea id="cr-body" placeholder="File contents…"></textarea>
        <div class="crow">
          <button class="abtn pri" type="button" id="cr-ok">Create file</button>
          <button class="abtn" type="button" id="cr-cancel">Cancel</button>
        </div>`;
    }
    box.classList.remove("hidden");
    const input = box.querySelector("#cr-name");
    setTimeout(() => input && input.focus(), 30);
    box.querySelector("#cr-cancel").onclick = () => { box.classList.add("hidden"); box.innerHTML = ""; };
    box.querySelector("#cr-ok").onclick = () => {
      const name = (box.querySelector("#cr-name").value || "").trim();
      if (!name) { input && input.focus(); return; }
      const ch = ensure(path);
      if (!ch) return;
      if (ch[name]) { alert("Already exists"); return; }
      if (kind === "folder") ch[name] = { type: "folder", children: {} };
      else ch[name] = { type: "file", content: (box.querySelector("#cr-body") && box.querySelector("#cr-body").value) || "" };
      if (path === "Desktop") paintIcons();
      paintFolder(body, path);
    };
    input && input.addEventListener("keydown", e => { if (e.key === "Enter") box.querySelector("#cr-ok").click(); });
  }

  function deleteItem(name) {
    if (!confirm('Delete "' + name + '"?')) return;
    const ch = ensure(path);
    if (!ch || !ch[name]) return;
    delete ch[name];
    if (path === "Desktop") paintIcons();
    const full = path + "/" + name;
    [...wins.entries()].forEach(([id, w]) => {
      if (w.path === full || (w.path && w.path.startsWith(full + "/"))) close(id);
    });
    paintFolder(body, path);
  }

  function renameItem(name) {
    box.innerHTML = `
      <label>Rename</label>
      <input type="text" id="cr-name" value="${esc(name)}" />
      <div class="crow">
        <button class="abtn pri" type="button" id="cr-ok">Rename</button>
        <button class="abtn" type="button" id="cr-cancel">Cancel</button>
      </div>`;
    box.classList.remove("hidden");
    const input = box.querySelector("#cr-name");
    setTimeout(() => { input.focus(); input.select(); }, 30);
    box.querySelector("#cr-cancel").onclick = () => { box.classList.add("hidden"); box.innerHTML = ""; };
    box.querySelector("#cr-ok").onclick = () => {
      const next = (input.value || "").trim();
      if (!next || next === name) { box.classList.add("hidden"); return; }
      const ch = ensure(path);
      if (!ch || !ch[name]) return;
      if (ch[next]) { alert("Already exists"); return; }
      ch[next] = ch[name];
      delete ch[name];
      const oldFull = path + "/" + name;
      const newFull = path + "/" + next;
      wins.forEach(w => {
        if (w.path === oldFull) {
          w.path = newFull;
          w.el.querySelector(".tt").textContent = next;
        } else if (w.path.startsWith(oldFull + "/")) {
          w.path = newFull + w.path.slice(oldFull.length);
        }
      });
      if (path === "Desktop") paintIcons();
      paintBar();
      paintFolder(body, path);
    };
    input.addEventListener("keydown", e => { if (e.key === "Enter") box.querySelector("#cr-ok").click(); });
  }

  function listFolderPaths(base, n, out) {
    if (!n || n.type !== "folder") return;
    out.push(base);
    Object.keys(n.children || {}).forEach(k => {
      if (n.children[k].type === "folder") listFolderPaths(base + "/" + k, n.children[k], out);
    });
  }

  function startMove(name) {
    selectedName = name;
    body.querySelectorAll(".fitem").forEach(el => {
      el.classList.toggle("selected", el.dataset.name === name);
    });
    const targets = [];
    listFolderPaths("Desktop", FS.Desktop, targets);
    const from = path + "/" + name;
    const sel = moveBar.querySelector("#move-target");
    sel.innerHTML = targets
      .filter(tp => tp !== path && !tp.startsWith(from + "/") && tp !== from)
      .map(tp => `<option value="${esc(tp)}">${esc(tp)}</option>`)
      .join("");
    moveBar.classList.add("show");
  }

  function doMove() {
    if (!selectedName) return;
    const dest = moveBar.querySelector("#move-target").value;
    if (!dest) return;
    const srcCh = ensure(path);
    const destCh = ensure(dest);
    if (!srcCh || !destCh || !srcCh[selectedName]) return;
    let name = selectedName;
    if (destCh[name]) {
      let n = 1;
      while (destCh[name + " (" + n + ")"]) n++;
      name = selectedName + " (" + n + ")";
    }
    destCh[name] = srcCh[selectedName];
    delete srcCh[selectedName];
    const oldFull = path + "/" + selectedName;
    const newFull = dest + "/" + name;
    wins.forEach(w => {
      if (w.path === oldFull) {
        w.path = newFull;
        w.el.querySelector(".tt").textContent = name;
      } else if (w.path.startsWith(oldFull + "/")) {
        w.path = newFull + w.path.slice(oldFull.length);
      }
    });
    if (path === "Desktop" || dest === "Desktop") paintIcons();
    paintBar();
    selectedName = null;
    moveBar.classList.remove("show");
    paintFolder(body, path);
  }

  // Tree navigation — open any folder/sub in THIS window
  body.querySelectorAll("[data-tree]").forEach(el => {
    el.onclick = () => navTo(el.dataset.tree);
  });

  // Breadcrumb
  body.querySelectorAll("[data-nav]").forEach(el => {
    el.onclick = () => navTo(el.dataset.nav);
  });

  // Items: single click select, double-click open in-place (folder) or new/read (file)
  body.querySelectorAll("[data-open]").forEach(el => {
    el.onclick = (e) => {
      if (e.target.closest(".acts")) return;
      body.querySelectorAll(".fitem").forEach(i => i.classList.remove("selected"));
      el.classList.add("selected");
      selectedName = el.dataset.name;
    };
    el.ondblclick = (e) => {
      if (e.target.closest(".acts")) return;
      const target = el.dataset.open;
      const n = node(target);
      if (!n) return;
      if (n.type === "folder") {
        // open subfolder in SAME window
        navTo(target);
      } else {
        // open file in a new window to read, keep folder open
        openPath(target);
      }
    };
  });

  body.querySelectorAll(".acts button").forEach(btn => {
    btn.onclick = (e) => {
      e.stopPropagation();
      const item = btn.closest(".fitem");
      const name = item && item.dataset.name;
      if (!name) return;
      const act = btn.dataset.act;
      if (act === "del") deleteItem(name);
      else if (act === "ren") renameItem(name);
      else if (act === "mov") startMove(name);
    };
  });

  body.querySelector('[data-act="nf"]').onclick = () => showCreate("folder");
  body.querySelector('[data-act="nfile"]').onclick = () => showCreate("file");
  body.querySelector('[data-act="up"]').onclick = () => {
    uploadPath = path;
    uploadBody = body;
    document.getElementById("uploader").click();
  };
  moveBar.querySelector("#move-go").onclick = doMove;
  moveBar.querySelector("#move-cancel").onclick = () => {
    selectedName = null;
    moveBar.classList.remove("show");
    body.querySelectorAll(".fitem").forEach(i => i.classList.remove("selected"));
  };
}

function paintFile(body, path, n) {
  const name = parts(path).pop();
  const parent = parts(path).slice(0, -1).join("/") || "Desktop";
  let content = "";
  if (n.type === "upload" && n.dataUrl && (n.mime || "").startsWith("image/")) {
    content = `<img src="${n.dataUrl}" alt="" />`;
  } else {
    content = `<pre>${esc(n.content || "")}</pre>`;
  }
  body.innerHTML = `<div class="filev">
    <div class="fbar" style="margin-bottom:10px">
      <button type="button" class="abtn" data-back="${parent}">← Back to folder</button>
      <button type="button" class="abtn" data-open-parent="${parent}">Open folder window</button>
    </div>
    <h3>${esc(name)}</h3>
    <div class="meta">${esc(path)}</div>
    ${content}
  </div>`;
  body.querySelector("[data-back]").onclick = () => {
    const winId = body.closest(".win").dataset.id;
    const w = wins.get(winId);
    if (!w) return;
    const parentPath = body.querySelector("[data-back]").dataset.back;
    const pn = node(parentPath);
    if (!pn || pn.type !== "folder") { openPath(parentPath); return; }
    w.path = parentPath;
    w.el.querySelector(".tt").textContent = parts(parentPath).pop();
    w.el.querySelector(".ic").textContent = "📁";
    paintFolder(body, parentPath);
    paintBar();
  };
  body.querySelector("[data-open-parent]").onclick = () => {
    openPath(body.querySelector("[data-open-parent]").dataset.openParent);
  };
}

function paintApp(body, app) {
  if (app === "search") {
    body.innerHTML = `<div class="ap"><h3>Search</h3>
      <input type="search" id="sq" placeholder="Search files & folders…" />
      <div id="sr"></div></div>`;
    const q = body.querySelector("#sq");
    const r = body.querySelector("#sr");
    q.oninput = () => {
      const t = q.value.trim().toLowerCase();
      if (!t) { r.innerHTML = ""; return; }
      const hits = [];
      (function walk(path, n) {
        if (n.type === "folder") Object.keys(n.children || {}).forEach(k => walk(path + "/" + k, n.children[k]));
        const name = parts(path).pop() || "";
        if (name.toLowerCase().includes(t) || (n.content && String(n.content).toLowerCase().includes(t))) {
          hits.push({ path, type: n.type, name });
        }
      })("Desktop", FS.Desktop);

/* ═══════════ Upload ═══════════ */
document.getElementById("uploader").onchange = async e => {
  const files = Array.from(e.target.files || []);
  e.target.value = "";
  if (!files.length) return;
  let path = uploadPath || "Desktop/Uploads";
  let ch = ensure(path);
  if (!ch) {
    const desk = ensure("Desktop");
    if (desk && !desk.Uploads) desk.Uploads = { type: "folder", children: {} };
    ch = ensure("Desktop/Uploads");
    path = "Desktop/Uploads";
  }
  if (!ch) return;
  for (const file of files) {
    let name = file.name;
    let n = 1;
    while (ch[name]) {
      const bits = file.name.split(".");
      const ext = bits.length > 1 ? "." + bits.pop() : "";
      name = `${bits.join(".") || "file"} (${n})${ext}`;
      n++;
    }
    const isText = /\.(txt|md|json|csv|html|css|js)$/i.test(file.name) || (file.type || "").startsWith("text/");
    if (isText && file.size < 500000) {
      ch[name] = { type: "file", content: await file.text() };
    } else if ((file.type || "").startsWith("image/") && file.size < 2e6) {
      const dataUrl = await new Promise(res => {
        const r = new FileReader();
        r.onload = () => res(r.result);
        r.readAsDataURL(file);
      });
      ch[name] = { type: "upload", mime: file.type, dataUrl, content: "[image]" };
    } else {
      ch[name] = { type: "upload", mime: file.type || "application/octet-stream", content: `[uploaded: ${file.name}, ${file.size} bytes]` };
    }
  }
  if (uploadBody && document.body.contains(uploadBody)) paintFolder(uploadBody, path);
  else openPath(path);
  if (path.startsWith("Desktop")) paintIcons();
  uploadBody = null;
};

/* ═══════════ Window ops ═══════════ */
function focus(id) {
  wins.forEach((w, wid) => w.el.classList.toggle("on", wid === id));
  const w = wins.get(id);
  if (w) {
    w.el.style.zIndex = ++z;
    if (w.min) { w.min = false; w.el.style.display = ""; }
  }
  active = id;
  paintBar();
}
function close(id) {
  const w = wins.get(id);
  if (!w) return;
  w.el.remove();
  wins.delete(id);
  if (active === id) active = null;
  paintBar();
}
function minimize(id) {
  const w = wins.get(id);
  if (!w) return;
  w.min = true;
  w.el.style.display = "none";
  w.el.classList.remove("on");
  if (active === id) active = null;
  paintBar();
}
function drag(win) {
  const bar = win.querySelector(".tb");
  let ox = 0, oy = 0, on = false;
  bar.onmousedown = e => {
    if (e.target.closest("button") || win.classList.contains("max")) return;
    on = true;
    ox = e.clientX - win.offsetLeft;
    oy = e.clientY - win.offsetTop;
    focus(win.dataset.id);
    e.preventDefault();
  };
  window.addEventListener("mousemove", e => {
    if (!on) return;
    win.style.left = Math.max(0, e.clientX - ox) + "px";
    win.style.top = Math.max(0, e.clientY - oy) + "px";
  });
  window.addEventListener("mouseup", () => { on = false; });
}
function resize(win) {
  const h = win.querySelector(".rh");
  let on = false, sx, sy, sw, sh;
  h.onmousedown = e => {
    if (win.classList.contains("max")) return;
    on = true; sx = e.clientX; sy = e.clientY;
    sw = win.offsetWidth; sh = win.offsetHeight;
    focus(win.dataset.id);
    e.preventDefault(); e.stopPropagation();
  };
  window.addEventListener("mousemove", e => {
    if (!on) return;
    win.style.width = Math.max(270, sw + e.clientX - sx) + "px";
    win.style.height = Math.max(170, sh + e.clientY - sy) + "px";
  });
  window.addEventListener("mouseup", () => { on = false; });
}
function paintBar() {
  const apps = document.getElementById("bar-apps");
  apps.innerHTML = "";
  wins.forEach((w, id) => {
    const name = parts(w.path).pop();
    const btn = document.createElement("button");
    btn.className = "open" + (id === active && !w.min ? " act" : "");
    btn.textContent = name;
    btn.onclick = () => {
      if (w.min || id !== active) focus(id);
      else minimize(id);
    };
    apps.appendChild(btn);
  });
}

  function applyDock(m) {
    mode = m;
    try { localStorage.setItem("ds_dock", m); } catch (e) {}
    bar.classList.remove("dock-bottom", "dock-top", "dock-left", "dock-right", "floating");
    document.body.classList.remove("bar-top", "bar-left", "bar-right");
    bar.style.left = "";
    bar.style.top = "";
    bar.style.right = "";
    bar.style.bottom = "";
    if (m === "float") {
      bar.classList.add("floating");
      if (!bar.style.left) {
        bar.style.left = "50%";
        bar.style.bottom = "16px";
        bar.style.transform = "translateX(-50%)";
      }
    } else {
      bar.style.transform = "";
      bar.classList.add("dock-" + m);
      if (m === "top") document.body.classList.add("bar-top");
      if (m === "left") document.body.classList.add("bar-left");
      if (m === "right") document.body.classList.add("bar-right");
    }
    menu.classList.remove("open");
  }

  applyDock(mode === "float" ? "float" : mode);

  grip.addEventListener("click", function (e) {
    e.stopPropagation();
    menu.classList.toggle("open");
  });
  menu.querySelectorAll("[data-dock]").forEach(btn => {
    btn.addEventListener("click", function (e) {
      e.stopPropagation();
      applyDock(this.dataset.dock);
    });
  });
  document.addEventListener("click", function (e) {
    if (!e.target.closest("#dock-menu") && !e.target.closest("#dock-grip")) {
      menu.classList.remove("open");
    }
  });

  // drag to float
  let dragging = false, ox = 0, oy = 0;
  grip.addEventListener("mousedown", function (e) {
    if (e.button !== 0) return;
    // only start drag after small move
    const sx = e.clientX, sy = e.clientY;
    function onMove(ev) {
      if (!dragging) {
        if (Math.abs(ev.clientX - sx) < 4 && Math.abs(ev.clientY - sy) < 4) return;
        dragging = true;
        menu.classList.remove("open");
        bar.classList.remove("dock-bottom", "dock-top", "dock-left", "dock-right");
        bar.classList.add("floating");
        document.body.classList.remove("bar-top", "bar-left", "bar-right");
        bar.style.transform = "";
        const rect = bar.getBoundingClientRect();
        bar.style.left = rect.left + "px";
        bar.style.top = rect.top + "px";
        bar.style.bottom = "auto";
        bar.style.right = "auto";
        ox = ev.clientX - rect.left;
        oy = ev.clientY - rect.top;
        mode = "float";
        try { localStorage.setItem("ds_dock", "float"); } catch (err) {}
      }
      bar.style.left = Math.max(0, Math.min(window.innerWidth - 80, ev.clientX - ox)) + "px";
      bar.style.top = Math.max(0, Math.min(window.innerHeight - 40, ev.clientY - oy)) + "px";
    }
    function onUp() {
      window.removeEventListener("mousemove", onMove);
      window.removeEventListener("mouseup", onUp);
      dragging = false;
    }
    window.addEventListener("mousemove", onMove);
    window.addEventListener("mouseup", onUp);
    e.preventDefault();
  });
})();

document.getElementById("pf-pic-input").addEventListener("change", function (e) {
  const file = e.target.files && e.target.files[0];
  e.target.value = "";
  if (!file || !file.type.startsWith("image/")) return;
  if (file.size > 1.5e6) { alert("Image too large (max ~1.5MB)"); return; }
  const reader = new FileReader();
  reader.onload = () => {
    avatarUrl = reader.result;
    try { localStorage.setItem("ds_avatar", avatarUrl); } catch (err) {}
    applyUser();
    // refresh open profile window if any
    for (const [id, w] of wins) {
      if (w.app === "profile" || w.path === "App/profile") {
        try { paintApp(w.el.querySelector(".body"), "profile"); } catch (err) {}
      }
    }
  };
  reader.readAsDataURL(file);
});


/* ═══════════ Desktop portrait (admin) ═══════════ */
function applyPortrait() {
  const art = document.getElementById("pic-art");
  if (!art) return;
  art.classList.remove("has-img", "has-video");
  if (portraitVideo) {
    art.classList.add("has-video");
    art.innerHTML = `<video src="${portraitVideo}" autoplay loop playsinline ${portraitMuted ? "muted" : ""}></video>
      <span class="vid-badge">${portraitMuted ? "MUTED" : "SOUND"}</span>`;
    const v = art.querySelector("video");
    if (v) {
      v.muted = !!portraitMuted;
      v.play().catch(() => {});
    }
  } else if (portraitUrl) {
    art.classList.add("has-img");
    art.innerHTML = `<img src="${portraitUrl}" alt="Admin portrait" />`;
  } else {
    art.innerHTML = `<svg viewBox="0 0 80 80" width="68" height="68" fill="none">
      <rect x="8" y="8" width="64" height="64" rx="14" fill="#0A0A0A" stroke="#C8A951" stroke-width="1.5"/>
      <text x="40" y="48" text-anchor="middle" font-family="Manrope,system-ui,sans-serif" font-weight="800" font-size="22" fill="#C8A951">MG</text>
    </svg>`;
  }
}
function setPortraitFromFile(file) {
  if (!file || !file.type.startsWith("image/")) return;
  if (file.size > 2e6) { alert("Image too large (max ~2MB)"); return; }
  const reader = new FileReader();
  reader.onload = () => {
    portraitUrl = reader.result;
    try { localStorage.setItem("ds_portrait", portraitUrl); } catch (e) {}
    applyPortrait();
  };
  reader.readAsDataURL(file);
}

document.getElementById("portrait-video-input").addEventListener("change", function (e) {
  const file = e.target.files && e.target.files[0];
  e.target.value = "";
  if (!file || !file.type.startsWith("video/")) return;
  if (file.size > 12e6) { alert("Video too large (max ~12MB for local save)"); return; }
  const reader = new FileReader();
  reader.onload = () => {
    portraitVideo = reader.result;
    try { localStorage.setItem("ds_portrait_video", portraitVideo); } catch (err) {

    }
    applyPortrait();
  };
  reader.readAsDataURL(file);
});

(function () {
  const pic = document.getElementById("pic");
  const ctx = document.getElementById("pic-ctx");
  if (!pic || !ctx) return;
  pic.addEventListener("contextmenu", function (e) {
    e.preventDefault();
    e.stopPropagation();
    ctx.style.left = Math.min(e.clientX, window.innerWidth - 190) + "px";
    ctx.style.top = Math.min(e.clientY, window.innerHeight - 140) + "px";
    ctx.classList.add("open");
  });
  document.addEventListener("click", function (e) {
    if (!e.target.closest("#pic-ctx")) ctx.classList.remove("open");
  });
  ctx.querySelectorAll("button").forEach(btn => {
    btn.addEventListener("click", function (e) {
      e.stopPropagation();
      const act = this.dataset.pact;
      ctx.classList.remove("open");
      if (act === "change") document.getElementById("portrait-input").click();
      else if (act === "video") document.getElementById("portrait-video-input").click();
      else if (act === "mute") {
        portraitMuted = !portraitMuted;
        try { localStorage.setItem("ds_portrait_muted", portraitMuted ? "1" : "0"); } catch (err) {}
        applyPortrait();
      } else if (act === "reset") {
        portraitUrl = null;
        portraitVideo = null;
        try {
          localStorage.removeItem("ds_portrait");
          localStorage.removeItem("ds_portrait_video");
        } catch (err) {}
        applyPortrait();
      } else if (act === "note") {
        const n = document.getElementById("pic-note");
        if (n) { n.focus(); }
      } else if (act === "bg") document.getElementById("bg-input").click();
      else if (act === "bg-reset") {
        desktopBg = null;
        try { localStorage.removeItem("ds_desktop_bg"); } catch (err) {}
        applyBackground();
      } else if (act === "open") openPath("Desktop/Portrait");
    });
  });
})();

  }
}
